    USERS {
        string id PK
        string fullName
        string email
        string provider
        string role
        boolean kycVerified
        datetime createdAt
    }

    LISTINGS {
        string id PK
        string title
        text description
        json pictures
        int numOfBeds
        decimal price
        boolean isFeatured
        decimal saleAmount
        date checkInDate
        date checkOutDate
        string hostId FK
        enum listingStatus
        enum locationType
        datetime createdAt
        datetime updatedAt
    }

    BOOKINGS {
        string id PK
        string listingId FK
        string guestId FK
        date checkInDate
        date checkOutDate
        enum status
        datetime createdAt
    }

    %% 关系定义
    USERS ||--o{ LISTINGS : "owns"
    USERS ||--o{ BOOKINGS : "makes"
    LISTINGS ||--o{ BOOKINGS : "has"
