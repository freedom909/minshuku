// Analytics Relationships
MATCH (g:Guest {id:"g1"}), (b:Booking {id:"b1"}) CREATE (g)-[:BOOKED]->(b);
MATCH (b:Booking {id:"b1"}), (l:Listing {id:"l1"}) CREATE (b)-[:FOR]->(l);

// Trend tagging

MATCH (l:Listing {id:"l2"}), (t:Trend {id:"t2"}) CREATE (l)-[:TAGGED_AS]->(t);
